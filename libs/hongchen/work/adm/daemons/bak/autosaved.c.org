// for autosave players data file.
#include <ansi.h>
#pragma save_binary

inherit F_DBASE;

void auto_save();
 
void create()
{
	seteuid(ROOT_UID);
	set("channel_id", "存盘精灵(AutoSave)");
	CHANNEL_D->do_channel( this_object(), "sys", "自动存盘系统已经启动。\n\n");
        call_out("auto_save", 40);
}

void auto_save()
{
        int i;
	object *ob, link_ob;
	string id;

	seteuid(getuid());
	ob=users();
	i=sizeof(ob);
        reclaim_objects();
	for( i=0;i<sizeof(ob);i++) {
		id = (string) ob[i]->query("id");
		if (!id) continue;
		if(!environment(ob[i]) )        continue;
		if( !objectp(link_ob = ob[i]->query_temp("link_ob")) ) continue;
		if( (int)link_ob->save() && (int)ob[i]->save() )
			message("channel:chat", "【"HIY"存盘精灵"NOR"】：您的档案已经自动存盘。\n"NOR,ob[i]);
		else
			message("channel:chat", "【"HIR"存盘精灵"NOR"】：您的档案自动存盘失败，请与巫师联系。\n"NOR,ob[i]);
	}
	remove_call_out("auto_save");
        call_out("auto_save", 600);
}
