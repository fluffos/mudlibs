//上边检测会发送一个ZJKEY的随机加密串给用户，用户接收到之后利用得到的加密串 str 连接指间mud网页网关获取新的加密串回复给游戏服务器
//服务器发送验证加密串 str 后等待用户回复，用户回复必须=crypt(ZJKEY,str[2..3])...此回复由用户调用指间MUD网关生成
void jiance(string arg,object ob,string str)
{
	if(!arg)
	{
		input_to("jiance",ob,str);
		return;
	}
	else if (arg==crypt(ZJKEY,str[2..3]))//如果接收到客户端回应是手机端
	{
		//版本验证成功:必须发送"版本验证成功\n"字串，客户端会对此字串做出回应，否则客户端会无响应
		write("版本验证成功\n");

		//客户端收到"版本验证成功\n"字串会自动发送指间网关的 "账号║密码║密文" 格式字串到游戏服务器，
		//密文验证方式为 crypt(ZJKEY,"账号")+crypt(ZJKEY,"密码"))=="密文"

		//游戏服务器可自行处理，如果要让客户用网关默认账号自动注册，可以仿照下面 get_user()函数适当修改
		//如果服务器不想让用户用网关账号自动注册，可以对网关账号数据不做处理，直接 write(SYSY"0007\n")
		//客户端会对 SYSY"0007\n" 字串做出回应跳转到主界面，此时主界面可以进行命令输入交互
		//验证完毕之后可以给予连接物件赋予是否手机登陆的变量，以便之后调用
		ob->set_temp("mob_login",1);
		input_to("get_user", ob);
		return;
	}
	else //如果需要屏蔽zmud，可以参考下边几行写法做出处理。如果要互通zmud下边可以转到zmud欢迎入口
	{
		write("客户端非法\n");
			log_file( "logins", sprintf("%s:%s\n",arg,str) );
		destruct(ob);
		return;
	}
}

//验证手机端账号密码函数
private void get_user(string arg, object ob)
{
	//arg,账号║密码║密文║email。
	object old_link,user;
	string *myinfo;

	if( !arg||arg=="") {
		input_to("get_user", ob);
		return;
	}

	myinfo = explode(arg,"║");

	if( sizeof(myinfo)!=4 ) {//数据串格式不对
		write(ZJTMPSAY"数据验证错误。。。\n");
		input_to("get_user", ob);
		return;
	}

	if( !check_legal_id(myinfo[0])) {
		//ID合法性检查，请自行相关修改，指间网关账号要求为字母打头的字母数字结合
		input_to("get_user", ob);
		return;
	}

	if( (crypt(ZJKEY,myinfo[0])+crypt(ZJKEY,myinfo[1]))!=myinfo[2] ) {
		write(ZJTMPSAY"账号数据校验错误，请重试。。。\n");
		input_to("get_user", ob);
		return;
	}

	myinfo[0] = lower_case(myinfo[0]);

	if( (string)ob->set("id", myinfo[0]) != myinfo[0]) {
		write(ZJTMPSAY"未知错误，请重试。。。\n");
		input_to("get_user", ob);
		return;
	}
	ob->set("email",myinfo[3]);
	if( file_size(ob->query_save_file() + __SAVE_EXTENSION__) >= 0 ) 
	{//如果账号存在，读取账号登陆数据
		if( !ob->restore() ) 
		{
			write(ZJTMPSAY"人物存档出现问题。。。\n");
			input_to("get_user", ob);
			return;
		}
		//此处可以加入账号如果账号已存在，2次密码验证
	}
	else 
		ob->set("body", USER_OB);//不存在，为新建账号做准备

	//不论账号是否是新建账号。根据网关账号更新游戏账号密码，担心安全问题请在上边加入2次验证
	if( (string)ob->set("password",crypt(myinfo[1],"zj")) != crypt(myinfo[1],"zj")) {
		write(ZJTMPSAY"未知错误，请重试。。。\n");
		input_to("get_user", ob);
		return;
	}

	write("\n");

	user = find_body(ob->query("id"));
	if (user) { //如果角色正在游戏中
		old_link = user->query_temp("link_ob");
		if( old_link ) {
			if(interactive(user))
			{
				tell_object(user, RED"你的账号在别处登录，你被迫下线了！\n");
				exec(old_link, user);
			}
			destruct(old_link);
		}
		//0007，登陆成功
		write(SYSY"0007\n");
		write(ZJTMPSAY"登录成功，正在加载世界。。。\n");
		reconnect(ob, user);
		return;
	}

	if( objectp(user = make_body(ob)) ) {
		user->set_temp("link_ob",ob);

		//如果角色数据存在，读取角色数据并正式进入游戏世界
		if( file_size(user->query_save_file() + __SAVE_EXTENSION__) >= 0 ){
			if( user->restore() ) {
				//////0007，登陆成功
				write(SYSY"0007\n");
				write(ZJTMPSAY"登录成功，正在加载世界。。。\n");
				enter_world(ob, user);
				return;
			} else {
				write(ZJTMPSAY"人物存档出现问题。。。\n");
				input_to("get_user", ob);
				return;
			}
		}
	}
	destruct(user);
	//角色数据不存在，软如新建角色处理，客户端在登陆界面中收到 SYSY"0008\n" 字串会打开建立角色界面，如果客户端已经打开命令交互，次项无效
	write(SYSY"0008\n");
	//角色界面包含 "性别║头像║昵称" 数据，确定建立会向服务器发送 "性别║头像║昵称"格式字串，可做如下处理 get_char()
	//如果
	input_to("get_char", ob);
}

private void get_char(string arg, object ob)
{
	//arg,性别║头像║昵称。
	object user;
	string *myinfo;

	if( !arg||arg=="") {
		input_to("get_char", ob);
		return;
	}

	myinfo = explode(arg,"║");
	if(sizeof(myinfo)!=3){ //数据串格式错误
		write(ZJTMPSAY"错误，请检查重试。\n");
		input_to("get_char", ob);
		return;
	}

	if( !check_legal_name(myinfo[2]) ){//昵称检查，服务器可根据情况自行设定检查
		//0009，昵称不符合要求
		input_to("get_char", ob);
		return;
	}

	ob->set("name", myinfo[2]);
	if( !objectp(user = make_body(ob)) )
		return;

//初始化角色，请根据MUD设定自行调整
	user->set("kar", 20 );
	user->set("str", 20 );
	user->set("cps", 20 );
	user->set("int", 25 );
	user->set("cor", 20 );
	user->set("con", 20 );
	user->set("spi", 25 );
	user->set("per", 20 );
	user->set("gender", myinfo[0]);
	user->set("img", myinfo[1]);

	user->set("no_gift",1);
	CHAR_D->setup_char(user); //setup user weight. mon 11/7/97
	user->set("title", "普通百姓");
	user->set("birthday", time() );
	user->set("potential",99);
	user->set("food", user->max_food_capacity());
	user->set("water", user->max_water_capacity());
	user->set("channels", ({ "chat","rumor","xyj","sldh","es" }) );
	log_file( "USAGE", sprintf("%s was created from %s (%s)\n", user->query("id"),
		query_ip_number(ob), ctime(time()) ) );

	ob->set("last_from",query_ip_number(ob));
	ob->set("last_on",time());
//0007，登陆成功。。客户端收到此消息会打开主界面窗口。主窗口可进行命令交互
	write(SYSY"0007\n");
	call_out("enter_world",1,ob, user);
}

#define ZJSEP		"$zj#"		//字串分隔符，比如向客户端发送场景中的多个npc信息，npc信息之间可用此分隔符连接

//定义此标志后文字的热点命令，点击此标志后直到 NOR 之前的文字 会触发 w 字串内包含的命令，
// w 字串格式："cmds:xxx"客户端会执行xxx命令，"http://xxxx","tel:xxx","mailto:xxx"会对应打开网页，电话，邮件程序
