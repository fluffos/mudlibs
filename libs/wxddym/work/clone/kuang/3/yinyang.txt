// yinyang.c 阴阳玉

#include <ansi.h>

inherit ITEM;

void create()
{
	set_name(HIM "阴阳玉" NOR, ({"magic stone", "yinyang yu", "yu"}));
	set_weight(6000);
	if (clonep())
		set_default_object(__FILE__);
	else {
		set("long", "三清道祖参悟阴阳之道，天地有感，孕育出和阴抱阳之玉。\n");
		set("unit", "块");
		set("value", 10000);
                set("item_origin", 1);
                set("material_attrib", "magic stone");
                set("material_name", HIM "神石" NOR);
                set("can_make", "all");
		set("not_make", ({ "战衣", "腰带" }));
                set("power_point", 340);
	}
}
