// hongmeng.c 鸿蒙石

#include <ansi.h>

inherit ITEM;

void create()
{
	set_name(HIM "鸿蒙石" NOR, ({"magic stone", "hongmeng shi", "shi"}));
	set_weight(6000);
	if (clonep())
		set_default_object(__FILE__);
	else {
		set("long", "鸿蒙，传说商户流传的奇物，染天地之灵，成九天造化。\n");
		set("unit", "块");
		set("value", 10000);
                set("item_origin", 1);
                set("material_attrib", "magic stone");
                set("material_name", HIM "神石" NOR);
                set("can_make", "all");
		set("not_make", ({ "战衣", "腰带" }));
                set("power_point", 350);
	}
}
