// jiulong.c 九龙震天

#include <ansi.h>

inherit ITEM;

void create()
{
	set_name(HIM "九龙镇天" NOR, ({"magic stone", "jiulong zhentian", "shi"}));
	set_weight(6000);
	if (clonep())
		set_default_object(__FILE__);
	else {
		set("long", "此神物乃石状，九龙环抱，散发出七彩之光，九龙之中隐含着天地之变化。\n");
		set("unit", "块");
		set("value", 10000);
                set("item_origin", 1);
                set("material_attrib", "magic stone");
                set("material_name", HIM "神石" NOR);
                set("can_make", "all");
		set("not_make", ({ "战衣", "腰带" }));
                set("power_point", 390);
	}
}
