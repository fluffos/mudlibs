// fengshen.c 封神石

#include <ansi.h>

inherit ITEM;

void create()
{
	set_name(HIM "封神石" NOR, ({"magic stone", "fengshen shi", "shi"}));
	set_weight(6000);
	if (clonep())
		set_default_object(__FILE__);
	else {
		set("long", "封神榜成之日渡天劫被第七道神雷劈下的一块石头。\n");
		set("unit", "块");
		set("value", 10000);
                set("item_origin", 1);
                set("material_attrib", "magic stone");
                set("material_name", HIM "神石" NOR);
                set("can_make", "all");
		set("not_make", ({ "战衣", "腰带" }));
                set("power_point", 350);
	}
}
