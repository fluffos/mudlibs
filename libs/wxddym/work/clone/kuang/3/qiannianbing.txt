// qiannianbing.c 千年玄冰

#include <ansi.h>

inherit ITEM;

void create()
{
	set_name(HIM "千年玄冰" NOR, ({"magic stone", "qiannian bing", "bing"}));
	set_weight(6000);
	if (clonep())
		set_default_object(__FILE__);
	else {
		set("long", "北冥无尽海海底产出的千年奇物。\n");
		set("unit", "块");
		set("value", 10000);
                set("item_origin", 1);
                set("material_attrib", "magic stone");
                set("material_name", HIM "神石" NOR);
                set("can_make", "all");
		set("not_make", ({ "战衣", "腰带" }));
                set("power_point", 320);
	}
}
