// putimu.c 菩提神木

#include <ansi.h>

inherit ITEM;

void create()
{
	set_name(HIM "菩提神木" NOR, ({"magic stone", "puti mu", "mu"}));
	set_weight(6000);
	if (clonep())
		set_default_object(__FILE__);
	else {
		set("long", "菩提本无树，明镜亦非台，本来无一物，何处惹尘埃。\n");
		set("unit", "块");
		set("value", 10000);
                set("item_origin", 1);
                set("material_attrib", "magic stone");
                set("material_name", HIM "神石" NOR);
                set("can_make", "all");
		set("not_make", ({ "战衣", "腰带" }));
                set("power_point", 390);
	}
}
