// jinghai.c 净海石

#include <ansi.h>

inherit ITEM;

void create()
{
	set_name(HIM "静海石" NOR, ({"magic stone", "jinghai shi", "shi"}));
	set_weight(6000);
	if (clonep())
		set_default_object(__FILE__);
	else {
		set("long", "静海阁阁主为情所困，丢下一颗静海石，只愿化情。\n");
		set("unit", "块");
		set("value", 10000);
                set("item_origin", 1);
                set("material_attrib", "magic stone");
                set("material_name", HIM "神石" NOR);
                set("can_make", "all");
		set("not_make", ({ "战衣", "腰带" }));
                set("power_point", 300);
	}
}
