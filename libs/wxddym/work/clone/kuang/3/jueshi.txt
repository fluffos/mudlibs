// jueshi.c 绝世

#include <ansi.h>

inherit ITEM;

void create()
{
	set_name(HIM "绝世" NOR, ({"magic stone", "jueshi shi", "shi"}));
	set_weight(6000);
	if (clonep())
		set_default_object(__FILE__);
	else {
		set("long", "干将隐退后因为这一件神材破了曾经的约定。\n");
		set("unit", "块");
		set("value", 10000);
                set("item_origin", 1);
                set("material_attrib", "magic stone");
                set("material_name", HIM "神石" NOR);
                set("can_make", "all");
		set("not_make", ({ "战衣", "腰带" }));
                set("power_point", 350);
	}
}
