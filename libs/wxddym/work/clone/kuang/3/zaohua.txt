// zaohua.c 造化神石

#include <ansi.h>

inherit ITEM;

void create()
{
	set_name(HIM "幽佛舍利" NOR, ({"magic stone", "zaohua shi", "shi"}));
	set_weight(6000);
	if (clonep())
		set_default_object(__FILE__);
	else {
		set("long", "没有人知道ta的来历，带在身上总有奇特的事情会发生。\n");
		set("unit", "颗");
		set("value", 10000);
                set("item_origin", 1);
                set("material_attrib", "magic stone");
                set("material_name", HIM "神石" NOR);
                set("can_make", "all");
		set("not_make", ({ "战衣", "腰带" }));
                set("power_point", 320);
	}
}
