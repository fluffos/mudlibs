// youfo.c 幽佛舍利

#include <ansi.h>

inherit ITEM;

void create()
{
	set_name(HIM "幽佛舍利" NOR, ({"magic stone", "youfo sheli", "sheli"}));
	set_weight(6000);
	if (clonep())
		set_default_object(__FILE__);
	else {
		set("long", "那修的九幽神佛转世后所留下的产物，幽光闪烁仿佛有老僧诉说着佛经。\n");
		set("unit", "颗");
		set("value", 10000);
                set("item_origin", 1);
                set("material_attrib", "magic stone");
                set("material_name", HIM "神石" NOR);
                set("can_make", "all");
		set("not_make", ({ "战衣", "腰带" }));
                set("power_point", 350);
	}
}
