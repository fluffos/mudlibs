// yaohuang.c 妖荒石

#include <ansi.h>

inherit ITEM;

void create()
{
	set_name(HIM "妖荒石" NOR, ({"magic stone", "yaohuang shi", "shi"}));
	set_weight(6000);
	if (clonep())
		set_default_object(__FILE__);
	else {
		set("long", "绝世妖王被裂天之炎焚化，一代天骄只落得一颗奇石而已。\n");
		set("unit", "块");
		set("value", 10000);
                set("item_origin", 1);
                set("material_attrib", "magic stone");
                set("material_name", HIM "神石" NOR);
                set("can_make", "all");
		set("not_make", ({ "战衣", "腰带" }));
                set("power_point", 360);
	}
}
