// yaoxing.c 妖星

#include <ansi.h>

inherit ITEM;

void create()
{
	set_name(HIM "妖星" NOR, ({"magic stone", "yaoxing shi", "shi"}));
	set_weight(6000);
	if (clonep())
		set_default_object(__FILE__);
	else {
		set("long", "昙花一现，正如这流星，带走的是美满的祝福，留下的却是抹不掉的思念和回忆，妖星！\n");
		set("unit", "颗");
		set("value", 10000);
                set("item_origin", 1);
                set("material_attrib", "magic stone");
                set("material_name", HIM "神石" NOR);
                set("can_make", "all");
		set("not_make", ({ "战衣", "腰带" }));
                set("power_point", 350);
	}
}
