#include <mudlib.h>


void create(object ob)

{


    if ( !ob ) return ;

	set_level(1);

	set_name( "fat farmer", "胖农夫" );

	add( "id", ({ "farmer" }) );

   	set_short( "胖农夫" );

	set_long(

		"这位农夫长得胖胖的，看起很老实。\n"

	);

	set( "max_hp", 40 );

	set( "hit_points", 40 );

	set( "race", "human" );

	set( "gender", "male" );

	set( "alignment", 200 );

	set( "wealth/silver", 15 );



	wield_weapon("/d/noden/item/fork");



	equip_armor("/d/noden/item/hat");

}
