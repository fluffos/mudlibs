// im create() Block des Gildenraumes, WT_XXXX sind die regulaeren 
// Weapon Types:
AddSkill( FIGHT(WT_KNIFE), ([ SI_SKILLFUNC: "Dolchkampf", 
    SI_DIFFICULTY: 100 ,SI_SKILLABILITY: 200]) );



// Hier ist die Funktion fuer den Dolchkampf:
mapping Dolchkampf(object player, string skill, mapping sinfo) 
{ 
  int abil, ud, val, bonus;

  ud=sinfo[SI_USR]; // Usage erhoehen
  abil = sinfo[SI_SKILLABILITY];
  // Berechnung des Bonus (immer prozentuell zu ability zu max_ability):
  val = (2* player->QueryAttribute(A_DEX) +  player->QueryAttribute(A_STR) +
         3*player->QueryProp(P_GUILD_LEVEL) + 2*player->QueryProp(P_GUILD_CIRCLE) 
        ) * abil / MAX_ABILITY;

  bonus=random(val);
  bonus=(bonus>MAX_KNIFE && interactive(player)?MAX_KNIFE:bonus);
  // Der Bonus kommt auf die Skilldamage drauf: 
  sinfo[SI_SKILLDAMAGE] += bonus * player->QuerySkillAttribute(SA_DAMAGE)/100;
  // Ab hier wird das "Lernen" ausgewertet. 
  // Wenn man Frosch ist und die Waffe weniger als ability/100 Damage macht 
  // -> kein lernen
  if( player->QueryProp(P_FROG) || 
      player->QueryProp(P_WEAPON)->QueryProp(P_WC) < abil/100 ) return sinfo;
  // Wenn der Gegner zuschwach ist (stufe) ist -> nicht lernen
  // Wir lernen auch erst, wenn wir den Skill oefter als ability/50 stel
  // angewandt haben. Damit lernt man langsamer je hoeher der Skill ist.
  // Wir lernen nur wenn die ability < max_ability ist
  // Ein bisserl Zufall: random 10 :) 
  if( (sinfo[SI_ENEMY]->QueryProp(P_LEVEL) > abil/400 || !random(10) )
     && ++ud > abil/50 && abil<MAX_ABILITY) 
  {
    ud = 0;
    abil += random(player->QuerProp(P_GUILD_LEVEL) + 
            player->QueryAttribute(A_DEX)/3 - GetAlign(player)/166);
    if (abil >= MAX_ABILITY)
      player->Message("Xarafi teilt Dir mit: Du kannst den Dolchkampf "
          "nun daemonisch!!!\n", MSGFLAG_TELL);
  }
  // Hier ein kleines Zuckerl :) 
  if(!random(500+MAX_ABILITY-abil) ) {
    tell_object(player, "Die Macht des Boesen verstaerkt Deine Attacke!\n");
    sinfo[SI_SKILLDAMAGE]+=400;
  }

  if(abil>MAX_ABILITY) abil=MAX_ABILITY;
  player->ModifySkill( FIGHT(WT_KNIFE), ([ SI_USR: ud, SI_SKILLABILITY: abil ]),
                       sinfo[SI_DIFFICULTY], "nekromanten");
  return sinfo;
}
