    return "";
   }

  if (i!=0)
   {write("Die Tuer zur Burgmauer steht offen.\n");
    return "";
   }

  return "";
  }
aufunc(str)
{object ob;


 if (!str)
 {notify_fail("So nicht!\n");
  return 0;
 }
 
 if (str=="tuer" || str == "tor" || str=="eingangstor")
  {if (!IS_WIZARD(TP) && !ob=present("\ntorkey",this_player()))
    {write ("Du brauchst den richtigen Schluessel um hier durchzukommen.\n");
     return 1;
    }

    write("Du steckst den Schuessel ins Schloss, sperrst auf und oeffnest\n"+
          "die Tuer. Durch magische Kraefte verschwindet der Schluessel.\n");
    say(this_player()->QueryProp(P_NAME)+" oeffnet eine Tuer.\n"); 
    if (!IS_WIZARD(TP)) ob->remove();
    i=1;
    return 1;
  }
 
  notify_fail("Was willst Du denn oeffnen?\n");
 return 0;
 }

 }
