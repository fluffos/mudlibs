void init() {
  ::init();
  object ob;
  int zufall;
  if( TP && interactive(TP) && TP->QueryProp(P_LEVEL) >= 5 && 
      TP->QueryProp(P_LEVEL) <= 10 && !present("wegelagerer") ) {
    zufall = random(500);
    if( zufall == 0 ) {
      AddItem(NPC + "wegelagerer.c", REFRESH_NONE);
      write(BS("Ein Wegelagerer springt Dir in den Weg!")
      ob = find_object("wegelagerer");
      ob -> attack(TP);
    }
  }
}
