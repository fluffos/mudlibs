Ich versuchs mal a bisserl zu erklaeren :-)

aufgerufen werden die funktionen mit
AMASTER->Funktion();


int QueryDefcon()
	Liefert einen integer Wert in welchem Status sich die Stadt
	befindet.
		DEFCON 0: Normalzustand
		DEFCON 1: NPCs wurden angegriffen
		DEFCON 2: Alarmzustand. Weniger als <x> NPCs sind
		          im Master registriert. Tore werden
		          geschlossen

	Beispiel:
	      if (AMASTER->QueryDefcon() == 2)
                  write("Die rote Alarmleuchte blinkt.\n");


int CheckDefcon()
	Es wird ueberprueft, ob die Voraussetzungen fuer DEFCON 2
	erfuellt sind und DERFCON 1 wird ausgeloest.
	Wird ausschliesslich von DelNpc aufgerufen.



int QueryDoorState(string tor)
	Liefert einen integerwert, ob ein Tor offen (1) oder
	geschlossen (0) ist.
	Gueltige Tore sind in def.h unter TORE definiert.
	Format in def.h: torname; status; filename, in dem das tor ist.
	in diesem wird beim Oeffnen des Tores OpenDoor() aufgerufen
	oder beim schliessen CloseDoor()
	Beispiel: 
               if (!AMASTER->QueryDoorState("Haupttor"))
	           write("Das Haupttor ist geschlossen!!\n");



int CloseDoor(string tor)
	Aendert einen Tuerstatus und ruft CloseDoor() in dem Raum in
	dem das Tor verzeichnet ist auf.
	Rueckgabewert ist 1 wenn die Tuer geschlossen wurde, und 0 wenn
	die Tuer bereits als geschlossen gespeichert war.
	Beispiel:
		if (AMASTER->CloseDoor())
		    write("Das Tor schliesst sich.\n");


int OpenDoor(string tor)
	Dasselbe wie CloseDoor :)


int CloseAllDoors()
	Alle Tore werden geschlossen. Wird von CheckDefcon() aufgerufen.


int OpenAllDoors()
	Alle Tore werden geoeffnet. Wird von CheckDefcon() aufgerufen.


int givelight()
	Liefert den stadtweiten Lichtlevel zurueck. Wird im STDROOM auf
	gerufen.


int givetemp()
	Siehe givelight()


int AddNpc(object npc)
	Speichert einen NPC im Master. Wird im STDNPC und STDMNPC im create()
	aufgerufen.


int DelNpc(object npc)
	Loescht einen NPC aus dem Master und loest ein CheckDefcon() aus.
	Wird von PlayerKilledNpc aufgerufen.


int PlayerKilledNpc(object npc, string player)
	Wird ein NPC getoetet, wird der Kill im Master fuer Freund-Feind-
	Erkennung gespeichert, und der NPC wird aus dem Master geloescht.
	Wird ausschliesslich von die() im STDNPC und STDMNPC aufgerufen.


int NpcKilledPlayer(string player)
	Speichert wenn ein NPC der Stadt einen Spieler toetet. Wird fuer
	Freund-Feind-Erkennung verwendet. Aufruf durch NotifyPlayerDeath
	im STDNPC und STDMNPC.
