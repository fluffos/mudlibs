int kletter (string str)
{
   string race;

   notify_fail ("Wohin willst Du klettern?\n");
   if (!str) return 0;
      
   race = TP->QueryProp(P_RACE);
   
    
   if ( str=="nach oben" || str=="oben" ||str=="rauf"||str=="in das dorf" || str=="in 
elfendorf"||str=="elfendorf"||str=="dorf"||str=="in dorf") 
   {
      switch(race)
      {
      case "Mensch":
      {
         write(BS("Du blickst noch einmal nach oben und "
            +"beginnst dann ohne Furch zu klettern."));
         TP->move(ROOM_DORF+"1_0914.c", M_GO, "klettert nach oben");
         return 1;
      }
      case "Zwerg":
      {
         write(BS("Nur mit groesster Ueberwindung beginnst Du "
            +"langsam und vorsichtig den Baum zu "
            +"ersteigen. Wenn Deine Ahnen das sehen "
            +"wuerden!"));
         TP->move(ROOM_DORF+"1_0914.c", M_GO, "klettert nach oben");
         return 1;
      }
      case "Hobbit":
      {
         write(BS("Frohen Mutes beginnst Du den riesigen Baum "
            +"zu erklettern. Obwohl es nicht leicht ist "
            +"fuer einen Hobbit, kommst Du gut voran."));
         TP->move(ROOM_DORF+"1_0914.c", M_GO, "klettert nach oben");
         return 1;
      }
      case "Elf":
      {
         write(BS("Froehlich steigst Du auf die ersten Aeste. "
            +"Hier fuehlst Du Dich zuhause."));
         TP->move(ROOM_DORF+"1_0914.c", M_GO, "klettert nach oben");
         return 1;
      }
      default:
      {
         write(BS("Du beginnst ohne Furcht nach oben zu klettern."));
         TP->move(ROOM_DORF+"1_0914.c", M_GO, "klettert nach oben");
         return 1;
      }
      }
   }
   return 0;
}
