// Pfluecken:
int cmdpfluecken(string str)
{
  object mybohne;

  notify_fail("Was willst Du pfluecken?\n");

  if(!str || str == "") return 0;

  if(!(str == "samen" ||
       str == "juckbohne" ||
       str == "juckbohnen" ||
       str == "bohne" ||
       str == "bohnen" )) return 0;

  if(i > 0)
  {
    mybohne = clone_object("/d/kristallwald/lilyth/waldpuck/obj/juckbohne");

    TP->Message(BS("Vorsichtig und mit spitzen Fingern pflueckst Du einen "
    +"besonders dicken Samen von der Pflanze."), MSGFLAG_VISUAL);

    TO->Message(BS(TP->Name(WER, 1)+" pflueckt etwas von einer seltsam "
    +"aussehenden Pflanze.\n"), MSGFLAG_VISUAL|MSGFLAG_NO_TP);

    if(ME_TOO_HEAVY == mybohne->move(TP,M_GET|M_SILENT))
    {
      mybohne->move(ENV(TP), M_PUT|M_SILENT);

      write(BS("Da Du den Samen nicht mehr tragen kannst, laesst Du "
      +"ihn einfach auf den Boden fallen."));

      ENV(TP)->Message(BS(TP->Name(WER, 1)+" laesst etwas zu Boden "
      +"fallen."), MSGFLAG_VISUAL|MSGFLAG_NO_TP);
      return -1;
     }

     i--;
     return 1;
  }

  if(i == 0)
  {
    TP->Message(BS("Es haengen nur noch ein paar vereinzelte, ziemlich "
    +"mickrig aussehende Samen an der Pflanze, die es sich kaum noch zu "
    +"pfluecken lohnt."), MSGFLAG_VISUAL);
    return -1;
  }
}
