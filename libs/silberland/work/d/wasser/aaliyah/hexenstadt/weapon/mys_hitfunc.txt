
int HitFunc(object enemy) 
{
  mapping resist;
  float water, fire, cold, air;

  if (!random(10) && sizeof(QueryProp(P_DAM_TYPE)) == 2 &&
      find_call_out("normal") == -1)
  {
    if (mappingp(resist = enemy->QueryProp(P_RESISTANCE_STRENGTHS)))
    {
      water = to_float(resist[DT_WATER]);
      fire = to_float(resist[DT_FIRE]);
      cold = to_float(resist[DT_COLD]);
      air = to_float(resist[DT_AIR]);

      if (water > 0.0 && water >= fire && water >= cold && water >= air)
      {
        SetProp(P_DAM_TYPE, QueryProp(P_DAM_TYPE) + ({DT_WATER}) );
        tell_object(QueryProp(P_WIELDED), 
            "Mysterium beschwoert eine Sturmflut.\n"); 
        QueryProp(P_WIELDED)->AddSp(-5 - random(5));
        call_out("normal", 6 + random(10) );
      }

      else if (fire > 0.0 && fire >= cold & fire >= air)
      {
        SetProp(P_DAM_TYPE, QueryProp(P_DAM_TYPE) + ({DT_FIRE}) );
        tell_object(QueryProp(P_WIELDED), 
            "Mysterium beschwoert einen Feuersturm.\n"); 
        QueryProp(P_WIELDED)->AddSp(-5 - random(5));
        call_out("normal", 6 + random(10) );
      }
      else if (cold > 0.0 && cold >= air)
      {
        SetProp(P_DAM_TYPE, QueryProp(P_DAM_TYPE) + ({DT_COLD}) );
        tell_object(QueryProp(P_WIELDED), 
            "Mysterium beschwoert einen Hagelschlag.\n"); 
        QueryProp(P_WIELDED)->AddSp(-5 - random(5));
        call_out("normal", 6 + random(10) );
      }
      else if (air > 0.0)
      {
        SetProp(P_DAM_TYPE, QueryProp(P_DAM_TYPE) + ({DT_AIR}) );
        tell_object(QueryProp(P_WIELDED), 
            "Mysterium beschwoert einen Wirbelsturm.\n"); 
        QueryProp(P_WIELDED)->AddSp(-5 - random(5));
        call_out("normal", 6 + random(10) );
      }
    }
  }
}
