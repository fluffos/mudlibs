string finger_date(string name)
{
	object ob, body;
	string msg, mud;
        int timeold,time,timecom;
	reset_eval_cost();
	if( sscanf(name, "%s@%s", name, mud)==2 ) {
		GFINGER_Q->send_gfinger_q(mud, name, this_player(1));
		return "网路指令传送过程可能需要一些时间，请稍候。\n";
	}

	ob = new(LOGIN_OB);
	ob->set("id", name);
	if( !ob->restore() ) {
	  destruct(ob);
	  return "没有这个玩家。\n";
        }
        timeold=ob->query("last_off");
        time=time();
        timecom=time-timeold;
	if( objectp(body = find_player(name)) 
		&& geteuid(body)==name 
		&& interactive(body)  ) 
	 {
	   msg = sprintf("目前正在连线中。\n");
         }
	else
	 {
	   msg = sprintf("距上次连线：\n");
	   msg = msg + age_string(timecom);
	 }  
	return msg;
}
