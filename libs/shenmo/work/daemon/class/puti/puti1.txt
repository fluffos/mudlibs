#include <ansi.h>
#include <combat.h>
inherit SSERVER;
int perform(object me, object target)
{
string msg;
int extra;
object weapon;
if(ob->is_busy())
                return notify_fail("你现在没空！！\n");
extra = ob->query_skill("unarmed");
if (extra = ob->query_skill("puti-zhi",1)<100)return notify_fail("你的菩提指还不够纯熟！\n");
if( !target
        ||      !target->is_character()
        ||      !ob->is_fighting(target) )
                return notify_fail("［菩提］只能对战斗中的对手使用。\n");
ob->add("force_factor",extra/5);
ob->set("force",extra/5*10);
ob->add_temp("apply/attack",extra/10);
weapon = ob->query_temp("weapon");
msg = HIY  "$N使出少林长拳中的［玉环鸳鸯拳］，一招连环七式向$n攻出！\n" NOR;
message_vision(msg,ob,target);
msg = HIY  "上一拳！" NOR;
COMBAT_D->do_attack(ob,target, ob->query_temp("weapon"),TYPE_REGULAR,msg);
msg = HIY  "下一拳！" NOR;
COMBAT_D->do_attack(ob,target, ob->query_temp("weapon"),TYPE_REGULAR,msg);
msg = HIY  "左一拳！" NOR;
COMBAT_D->do_attack(ob,target, ob->query_temp("weapon"),TYPE_REGULAR,msg);
msg = HIY  "右一拳！" NOR;
COMBAT_D->do_attack(ob,target, ob->query_temp("weapon"),TYPE_REGULAR,msg);
msg = HIY  "前一拳！" NOR;
COMBAT_D->do_attack(ob,target, ob->query_temp("weapon"),TYPE_REGULAR,msg);
msg = HIY  "后一拳！" NOR;
COMBAT_D->do_attack(ob,target, ob->query_temp("weapon"),TYPE_REGULAR,msg);
msg = HIY  "最后再一拳！！" NOR;
COMBAT_D->do_attack(ob,target, ob->query_temp("weapon"),TYPE_REGULAR,msg);
ob->start_busy(5);
ob->set("force_factor",0);
ob->set("force",0);
ob->add_temp("apply/attack",-extra/10);
return 1;
}
