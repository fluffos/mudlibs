#include <armor.h>

inherit {inherit};
inherit F_ITEMMAKE;

void create()
{
    set_name("{name}", ({ "{id}" }));
    set_weight({weight});
    set("item_make", 1);
    set("unit", "{unit}");
    set("long", "{long}");
    set("value", {value});
    set("point", {point});
    set("material", "{material}");
    set("wear_msg", "{wear_msg}\n");
    set("remove_msg", "{remove_msg}\n");
    if (! check_clone()) return;
    restore();
    {armor_prop}
    setup();
}
