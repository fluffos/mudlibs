#include <weapon.h>

inherit {inherit};
inherit F_ITEMMAKE;

void create()
{
    set_name("{name}", ({ "{id}" }));
    set_weight({weight});
    set("item_make", 1);
    set("unit", "{unit}");
    set("long", "{long}");
    set("value", {value});
    set("point", {point});
    set("material", "{material}");
    set("wield_msg", "{wield_msg}\n");
    set("unwield_msg", "{unwield_msg}\n");
    if (! check_clone()) return;
    restore();
    init_{stype}(apply_damage());
    {prop_segment}
    setup();
}
