<ClassBrowser language="lpc" name="LOOT_D" source="adm/daemons/loot.c" type="object">
  Daemon responsible for handling the loot system, including item drops,
  coin drops, and automatic valuation. Called by the death sequence when
  an NPC is killed.

  <ClassExtends label="inherits" name="STD_DAEMON" source="std/daemon/daemon.c" highlight="object" target="/systems/loot/" />
  <ClassExtends label="inherits" name="EXT_LOG" source="std/ext/log.c" highlight="object" target="/systems/loot/" />

  <ClassProperty name="log_level" label="variable" highlight="variable">
    <ClassPropertyType type="int" highlight="type" />
  </ClassProperty>

  <ClassMethod name="loot_drop" visibility="public" highlight="function">
    Drops loot from the source's loot table into the source (corpse).
    Iterates each entry, rolls against its chance, and creates the item
    on success.

    <ClassMethodInput name="tp" type="STD_BODY" highlight="parameter" hint="std/body.c" target="/systems/loot/">The player who killed the NPC</ClassMethodInput>
    <ClassMethodInput name="source" type="STD_NPC" highlight="parameter" hint="std/living/npc.c" target="/systems/loot/">The NPC whose loot table to process</ClassMethodInput>
    <ClassMethodOutput type="void" highlight="type" />
  </ClassMethod>

  <ClassMethod name="coin_drop" visibility="public" highlight="function">
    Drops coins from the source's coin table into the source. Each coin
    entry is rolled independently.

    <ClassMethodInput name="tp" type="STD_BODY" highlight="parameter">The player who killed the NPC</ClassMethodInput>
    <ClassMethodInput name="source" type="STD_NPC" highlight="parameter">The NPC whose coin table to process</ClassMethodInput>
    <ClassMethodOutput type="void" highlight="type" />
  </ClassMethod>

  <ClassMethod name="determine_value_by_level" visibility="public nomask" highlight="function">
    Calculates a coin value based on level, applying a configured variance
    for randomisation. Uses `COIN_VALUE_PER_LEVEL` and `COIN_VARIANCE`
    from the MUD configuration.

    <ClassMethodInput name="level" type="int" highlight="parameter">The NPC's level</ClassMethodInput>
    <ClassMethodOutput type="int" highlight="type">The calculated coin value</ClassMethodOutput>
  </ClassMethod>

  <ClassMethod name="validate_loot_entry" visibility="protected" highlight="function">
    Validates a loot table entry before processing. Checks that the item
    reference is well-formed and the chance is within range.

    <ClassMethodInput name="entry" type="mixed" highlight="parameter">The loot table entry to validate</ClassMethodInput>
    <ClassMethodOutput type="int" highlight="type">1 if valid, 0 otherwise</ClassMethodOutput>
  </ClassMethod>

  <ClassMethod name="drop_items" visibility="private" highlight="function">
    Processes and drops a single loot item into the source. Resolves the
    item via `process_loot_item`, creates the object, auto-values it if
    applicable, and moves it into the corpse.

    <ClassMethodInput name="tp" type="STD_BODY" highlight="parameter">The player</ClassMethodInput>
    <ClassMethodInput name="item" type="mixed" highlight="parameter">The loot table entry to resolve</ClassMethodInput>
    <ClassMethodInput name="source" type="STD_NPC" highlight="parameter">The NPC to drop into</ClassMethodInput>
    <ClassMethodOutput type="void" highlight="type" />
  </ClassMethod>

  <ClassMethod name="process_loot_item" visibility="private" highlight="function">
    Recursively resolves a loot item. Handles functions (called and
    re-resolved), arrays (random pick), weighted mappings
    (`element_of_weighted`), and plain string paths.

    <ClassMethodInput name="item" type="mixed" highlight="parameter">The item to resolve — string path, mapping, array, or function</ClassMethodInput>
    <ClassMethodInput name="tp" type="STD_BODY" highlight="parameter">The player</ClassMethodInput>
    <ClassMethodInput name="source" type="STD_NPC" highlight="parameter">The NPC source</ClassMethodInput>
    <ClassMethodOutput type="mixed" highlight="type">A file path string, a `({ path, args })` pair, or `0` on failure</ClassMethodOutput>
  </ClassMethod>

  <ClassMethod name="autovalue_loot_item" visibility="private" highlight="function">
    Sets the value of a loot item based on the source NPC's level, if
    the item has the `autovalue` loot property enabled.

    <ClassMethodInput name="item" type="OBJ_LOOT" highlight="parameter">The loot item to value</ClassMethodInput>
    <ClassMethodInput name="tp" type="STD_BODY" highlight="parameter">The player</ClassMethodInput>
    <ClassMethodInput name="source" type="STD_NPC" highlight="parameter">The NPC whose level determines value</ClassMethodInput>
    <ClassMethodOutput type="void" highlight="type" />
  </ClassMethod>

  <ClassMethod name="drop_coins" visibility="private" highlight="function">
    Creates a coin object of the given type and amount, and moves it
    into the source.

    <ClassMethodInput name="tp" type="STD_BODY" highlight="parameter">The player</ClassMethodInput>
    <ClassMethodInput name="item" type="mixed" highlight="parameter">A `({ currency_type, amount })` pair</ClassMethodInput>
    <ClassMethodInput name="source" type="STD_NPC" highlight="parameter">The NPC to drop coins into</ClassMethodInput>
    <ClassMethodOutput type="void" highlight="type" />
  </ClassMethod>
</ClassBrowser>
