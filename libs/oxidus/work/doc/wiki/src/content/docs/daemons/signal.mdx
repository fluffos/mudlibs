<ClassBrowser language="lpc" name="SIGNAL_D" source="adm/daemons/signal.c" type="object">
  Daemon providing a system-wide event notification system. Manages
  registration, dispatch, and cleanup of signal handlers (slots), and
  persists them across reboots via the swap daemon. Every function below
  rejects callers other than the simul_efun layer — register, emit, and
  unregister through the `slot()`, `emit()`, and `unslot()` simul_efuns.

  <ClassExtends label="inherits" name="STD_DAEMON" source="std/daemon/daemon.c" highlight="object" target="/systems/signal/" />

  <ClassProperty name="slots" label="variable" highlight="variable">
    <ClassPropertyType type="mapping" highlight="type" />
  </ClassProperty>

  <ClassMethod name="register_slot" visibility="public nomask" highlight="function">
    Registers an object's function to receive a specific signal. One slot
    per object per signal — a second registration overwrites the first.
    Rejected unless the caller is the simul_efun layer.

    <ClassMethodInput name="sig" type="string" highlight="parameter">The signal identifier to register for</ClassMethodInput>
    <ClassMethodInput name="ob" type="object" highlight="parameter">The object to receive the signal</ClassMethodInput>
    <ClassMethodInput name="func" type="string" highlight="parameter">The function to call when the signal fires</ClassMethodInput>
    <ClassMethodOutput type="int" highlight="type">`SIG_SLOT_OK` on success, or a negative `SIG_*` error code</ClassMethodOutput>
  </ClassMethod>

  <ClassMethod name="unregister_slot" visibility="public nomask" highlight="function">
    Unregisters an object from a signal. Removes the signal entry entirely
    once its last handler is gone. Rejected unless the caller is the
    simul_efun layer.

    <ClassMethodInput name="sig" type="string" highlight="parameter">The signal identifier to unregister from</ClassMethodInput>
    <ClassMethodInput name="ob" type="object" highlight="parameter">The object to unregister</ClassMethodInput>
    <ClassMethodOutput type="int" highlight="type">`SIG_SLOT_OK` on success, or a negative `SIG_*` error code</ClassMethodOutput>
  </ClassMethod>

  <ClassMethod name="dispatch_signal" visibility="public nomask" highlight="function">
    Broadcasts a signal to every registered handler. Each call is wrapped
    in `catch()` — a failing handler is logged to the `SIGNAL_ERROR` log and
    never blocks the others. Fire-and-forget: no return value. Rejected
    unless the caller is the simul_efun layer.

    <ClassMethodInput name="sig" type="string" highlight="parameter">The signal identifier to dispatch</ClassMethodInput>
    <ClassMethodInput name="arg..." type="mixed" highlight="parameter">Arguments passed through to each handler</ClassMethodInput>
    <ClassMethodOutput type="void" highlight="type" />
  </ClassMethod>

  <ClassMethod name="invalidate_slots" visibility="private nomask" highlight="function">
    Prunes invalid slots — those whose object has been destructed or whose
    handler function no longer exists. Runs on startup, on every heartbeat
    (60s), and once more before shutdown.

    <ClassMethodOutput type="void" highlight="type" />
  </ClassMethod>
