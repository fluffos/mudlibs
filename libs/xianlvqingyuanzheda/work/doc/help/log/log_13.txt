指令格式 : unique <法宝中文名称> | all                                                       
这个指令告诉你所指定的法宝是否已经被玩家持有。                                               
如果使用参数all则列出所有法宝状态。                                                          
以下为法宝中文名称列表:                                                                      
=======================================                                                      
火灵珠                                                                                       
圣灵珠                                                                                       
金箍棒                                                                                       
风灵珠                                                                                       
水灵珠                                                                                       
旧白布搭包                                                                                   
土灵珠                                                                                       
雷灵珠                                                                                       
=======================================                                                      


Code:                                                                                        


//written by bugbug 2002-3-10                                                                
#include <ansi.h>                                                                            
inherit F_CLEAN_UP;                                                                          
int report(object me,string name);                                                           
int help(object me);                                                                         
mapping name_location=                                                                       
([                                                                                           
 "风灵珠":         "/d/obj/baowu/fengzhu",                                                   
 "水灵珠":         "/d/obj/baowu/shuizhu",                                                   
 "火灵珠":         "/d/obj/baowu/huozhu",                                                    
 "雷灵珠":         "/d/obj/baowu/leizhu",                                                    
 "圣灵珠":         "/d/obj/baowu/shengzhu",                                                  
 "土灵珠":         "/d/obj/baowu/tuzhu",                                                     
 "金箍棒":         "/d/dntg/donghai/obj/jingubang",                                          
 "旧白布搭包":     "/d/qujing/xiaoxitian/obj/baoreal",                                       
]);                                                                                          
int main(object me, string arg)                                                              
{                                                                                            
 string* name=keys(name_location);                                                           
 if(!arg)                                                                                    
 {                                                                                           
  help(me);                                                                                  
  return 1;                                                                                  
 }                                                                                           
 if( arg =="all" )                                                                           
 {                                                                                           
  tell_object(me,"\n");                                                                      
  foreach(string temp in name)                                                               
   report(me,temp);                                                                          
  return 1;                                                                                  
 }                                                                                           
                                                                                                                       
 report(me,arg);                                                                             
 return 1;                                                                                   
}                                                                                            
int report(object me,string name)                                                            
{                                                                                            
 string location,msg;                                                                        
 if( undefinedp(location=name_location[name]) )                                              
 {                                                                                           
  help(me);                                                                                  
  return 1;                                                                                  
 }                                                                                           
 msg = location->in_mud() ? HIR "已经有人" NOR : HIC "还没有人" NOR ;                        
 msg+= "拥有"+name+"。\n";                                                                   
 tell_object(me,msg);                                                                        
 return 1;                                                                                   
}                                                                                            
int help(object me)                                                                          
{                                                                                            
 string* name=keys(name_location);                                                           
 write(@HELP

 write("=======================================\n");
