#include <ansi.h>

inherit F_SSERVER;
int perform(object me, object target)
{
	string msg;

	if( !target ) target = offensive_target(me);

	if( !target
	||	!target->is_character()
	||	!me->is_fighting(target) )
		return notify_fail("天旋地转只能对战斗中的对手使用。\n");

	if( target->is_busy() )
		return notify_fail(target->name() +已经手忙脚乱，失去了还手之力。 "ⅵ\n");
		
	if( (int)me->query_skill("qixing-huanwei", 1) < 120 )
		return notify_fail("你的七星换位不够娴熟，不会使用天旋地转。\n");

	msg = CYN "$N使出七星换位的「天旋地转」，脚踩连环，绕着$n转个不停。\n";
      
       		
	me->start_busy(1);
	if( random(me->query("combat_exp")) > (int)target->query("combat_exp")/2 ) {
		msg += "结果$p被$P转的眼花缭乱。ⅵ\n" NOR;
		target->start_busy( (int)me->query_skill("taiji-jian") / 20 );
	} else {
		msg += "结果$P自己转得眼花缭乱。\n" NOR;
	}
	message_vision(msg, me, target);

	return 1;
}
